-- ============================================
-- 商品分类表
-- ============================================
CREATE TABLE IF NOT EXISTS product_categories (
    id TEXT PRIMARY KEY NOT NULL,
    name TEXT NOT NULL,
    parent_id TEXT,
    level INTEGER NOT NULL CHECK (level IN (1, 2)),
    sort_order INTEGER DEFAULT 0,
    created_at TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at TEXT NOT NULL DEFAULT (datetime('now')),
    FOREIGN KEY (parent_id) REFERENCES product_categories(id) ON DELETE SET NULL
);

-- 分类层级索引
CREATE INDEX IF NOT EXISTS idx_category_level ON product_categories(level);
CREATE INDEX IF NOT EXISTS idx_category_parent ON product_categories(parent_id);

-- ============================================
-- 商品表
-- ============================================
CREATE TABLE IF NOT EXISTS products (
    id TEXT PRIMARY KEY NOT NULL,
    code TEXT NOT NULL UNIQUE,
    name TEXT NOT NULL,
    category_id TEXT,
    type TEXT NOT NULL CHECK (type IN ('weight', 'count')),
    base_unit TEXT NOT NULL CHECK (base_unit IN ('g', 'pcs')),
    origin TEXT,
    year TEXT,
    grade TEXT,
    fermentation_level TEXT,
    roast_level TEXT,
    image_url TEXT,
    default_unit_id TEXT,
    is_active INTEGER NOT NULL DEFAULT 1,
    created_at TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at TEXT NOT NULL DEFAULT (datetime('now')),
    FOREIGN KEY (category_id) REFERENCES product_categories(id) ON DELETE SET NULL
);

-- 商品编码唯一索引
CREATE UNIQUE INDEX IF NOT EXISTS idx_product_code ON products(code);
-- 分类索引
CREATE INDEX IF NOT EXISTS idx_product_category ON products(category_id);
-- 类型索引
CREATE INDEX IF NOT EXISTS idx_product_type ON products(type);
-- 名称搜索索引（支持模糊查询）
CREATE INDEX IF NOT EXISTS idx_product_name ON products(name);

-- ============================================
-- 销售单位表
-- ============================================
CREATE TABLE IF NOT EXISTS sales_units (
    id TEXT PRIMARY KEY NOT NULL,
    product_id TEXT NOT NULL,
    name TEXT NOT NULL,
    conversion_to_base INTEGER NOT NULL CHECK (conversion_to_base > 0),
    retail_price REAL NOT NULL CHECK (retail_price >= 0),
    member_price REAL NOT NULL CHECK (member_price >= 0),
    sort_order INTEGER DEFAULT 0,
    created_at TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at TEXT NOT NULL DEFAULT (datetime('now')),
    FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE
);

-- 商品索引（级联删除）
CREATE INDEX IF NOT EXISTS idx_unit_product ON sales_units(product_id);

## 四、核心逻辑

### 4.1 商品编码自动生成

```rust
/// 生成商品编码
/// 格式：SP + 年月日 + 3位序号
/// 示例：SP20260701001
fn generate_product_code(db: &Connection) -> Result<String, String> {
    let today = chrono::Local::now().format("%Y%m%d").to_string();
    let prefix = format!("SP{}", today);
    
    // 查找今天最大的序号
    let max_code: Option<String> = db.query_row(
        "SELECT MAX(code) FROM products WHERE code LIKE ?",
        [&format!("{}%", prefix)],
        |row| row.get(0)
    ).ok();
    
    let sequence = match max_code {
        Some(code) => {
            let num: u32 = code[prefix.len()..].parse().unwrap_or(0);
            num + 1
        }
        None => 1,
    };
    
    Ok(format!("{}{:03}", prefix, sequence))
}
```

### 4.2 基准单位自动设置

```rust
/// 根据商品类型设置基准单位
fn get_base_unit(product_type: &ProductType) -> BaseUnit {
    match product_type {
        ProductType::Weight => BaseUnit::Gram,   // 称重类 → 克
        ProductType::Count => BaseUnit::Pieces,  // 计件类 → 个
    }
}
```

### 4.3 创建商品事务

```rust
/// 创建商品（带事务）
pub fn create_product_with_units(
    db: &Connection,
    input: &ProductInput,
) -> Result<Product, String> {
    // 生成编码
    let code = generate_product_code(db)?;
    let base_unit = get_base_unit(&input.product_type);
    
    // 生成商品ID
    let product_id = uuid::Uuid::new_v4().to_string();
    let now = chrono::Local::now().to_rfc3339();
    
    // 开启事务
    db.execute("BEGIN TRANSACTION", [])
        .map_err(|e| e.to_string())?;
    
    try {
        // 插入商品
        db.execute(
            "INSERT INTO products (id, code, name, category_id, type, base_unit, 
             origin, year, grade, fermentation_level, roast_level, image_url, 
             is_active, created_at, updated_at)
             VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 1, ?, ?)",
            rusqlite::params![
                product_id,
                code,
                input.name,
                input.category_id,
                serde_json::to_string(&input.product_type).unwrap(),
                serde_json::to_string(&base_unit).unwrap(),
                input.origin,
                input.year,
                input.grade,
                input.fermentation_level,
                input.roast_level,
                input.image_url,
                now,
                now
            ],
        ).map_err(|e| e.to_string())?;
        
        // 插入销售单位
        let mut default_unit_id = None;
        for (idx, unit) in input.units.iter().enumerate() {
            let unit_id = unit.id.clone().unwrap_or_else(|| uuid::Uuid::new_v4().to_string());
            
            db.execute(
                "INSERT INTO sales_units (id, product_id, name, conversion_to_base, 
                 retail_price, member_price, sort_order, created_at, updated_at)
                 VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
                rusqlite::params![
                    unit_id,
                    product_id,
                    unit.name,
                    unit.conversion_to_base,
                    unit.retail_price,
                    unit.member_price,
                    idx as i32,
                    now,
                    now
                ],
            ).map_err(|e| e.to_string())?;
            
            // 第一个单位设为默认
            if idx == 0 {
                default_unit_id = Some(unit_id.clone());
            }
        }
        
        // 更新默认单位
        if let Some(ref unit_id) = default_unit_id {
            db.execute(
                "UPDATE products SET default_unit_id = ? WHERE id = ?",
                rusqlite::params![unit_id, product_id],
            ).map_err(|e| e.to_string())?;
        }
        
        db.execute("COMMIT", []).map_err(|e| e.to_string())?;
        
        Ok(Product { id: product_id, code, name: input.name.clone(), /* ... */ })
    } catch {
        |e| {
            db.execute("ROLLBACK", []).map_err(|_| e.to_string())?;
            Err(e.to_string())
        }
    }
}
```

```vue
<!-- 销售单位编辑器 -->
<template>
  <div class="sales-unit-editor">
    <div v-for="(unit, index) in units" :key="index" class="unit-row">
      <n-input v-model="unit.name" placeholder="单位名称" />
      <n-input-number v-model="unit.conversionToBase" :min="1" />
      <span>{{ baseUnitText }}</span>
      <n-input-number v-model="unit.retailPrice" :min="0" :precision="2" />
      <n-input-number v-model="unit.memberPrice" :min="0" :precision="2" />
      <n-button @click="removeUnit(index)">删除</n-button>
    </div>
    <n-button @click="addUnit">添加单位</n-button>
  </div>
</template>

<script setup lang="ts">
import type { SalesUnitInput } from '@/types'

const props = defineProps<{
  modelValue: SalesUnitInput[]
  baseUnit: 'g' | 'pcs'
}>()

const emit = defineEmits<{
  (e: 'update:modelValue', value: SalesUnitInput[]): void
}>()

const baseUnitText = computed(() => props.baseUnit === 'g' ? '克' : '个')

function addUnit() {
  emit('update:modelValue', [
    ...props.modelValue,
    { name: '', conversionToBase: 1, retailPrice: 0, memberPrice: 0 }
  ])
}

function removeUnit(index: number) {
  emit('update:modelValue', props.modelValue.filter((_, i) => i !== index))
}
</script>
```

## 七、风险与应对

| 风险 | 影响 | 应对措施 |
|------|------|---------|
| 事务执行失败 | 数据不一致 | 使用 SQLite 事务，失败自动回滚 |
| 并发创建商品编码重复 | 编码唯一性 | 使用事务 + 锁保证序号唯一 |
| 商品类型切换导致库存混乱 | 业务逻辑错误 | 有库存记录的商品禁止修改类型 |
