## 三、状态管理

### 3.1 `stores/tabs.ts`

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export interface Tab {
  key: string           // 路由 path
  title: string
  closable: boolean
}

const STORAGE_KEY = 'tea-tabs'
const HOME_TAB: Tab = { key: '/', title: '首页', closable: false }

function loadFromStorage(): { tabs: Tab[]; activeKey: string } {
  try {
    const saved = localStorage.getItem(STORAGE_KEY)
    if (saved) {
      const parsed = JSON.parse(saved)
      if (parsed.tabs?.length > 0) {
        const hasHome = parsed.tabs.find((t: Tab) => t.key === '/')
        if (!hasHome) parsed.tabs.unshift({ ...HOME_TAB })
        if (!parsed.tabs.find((t: Tab) => t.key === parsed.activeKey)) {
          parsed.activeKey = '/'
        }
        return { tabs: parsed.tabs, activeKey: parsed.activeKey }
      }
    }
  } catch {}
  return { tabs: [{ ...HOME_TAB }], activeKey: '/' }
}

function saveToStorage(tabs: Tab[], activeKey: string) {
  localStorage.setItem(STORAGE_KEY, JSON.stringify({ tabs, activeKey }))
}

export const useTabsStore = defineStore('tabs', () => {
  const initial = loadFromStorage()
  const tabs = ref<Tab[]>(initial.tabs)
  const activeKey = ref<string>(initial.activeKey)
  const dragIndex = ref<number | null>(null)
  const dragOverIndex = ref<number | null>(null)

  const activeTab = computed(() => tabs.value.find(t => t.key === activeKey.value))

  function addTab(tab: Tab) {
    const exists = tabs.value.find(t => t.key === tab.key)
    if (exists) {
      activeKey.value = tab.key
    } else {
      tabs.value.push(tab)
      activeKey.value = tab.key
    }
    persist()
  }

  function removeTab(key: string) {
    const tab = tabs.value.find(t => t.key === key)
    if (tab?.closable === false) return
    const idx = tabs.value.findIndex(t => t.key === key)
    if (idx === -1) return
    tabs.value.splice(idx, 1)
    if (tabs.value.length === 0) {
      tabs.value.push({ ...HOME_TAB })
    }
    if (activeKey.value === key) {
      const nextIdx = idx < tabs.value.length ? idx : tabs.value.length - 1
      activeKey.value = tabs.value[nextIdx].key
    }
    persist()
  }

  function setActiveKey(key: string) {
    activeKey.value = key
    persist()
  }

  function closeOtherTabs(key: string) {
    tabs.value = tabs.value.filter(t => t.key === key || t.closable === false)
    activeKey.value = key
    persist()
  }

  function closeAllTabs() {
    tabs.value = tabs.value.filter(t => t.closable === false)
    if (tabs.value.length > 0) activeKey.value = tabs.value[0].key
    persist()
  }

  function closeLeftTabs(key: string) {
    const idx = tabs.value.findIndex(t => t.key === key)
    if (idx <= 0) return
    const newTabs: Tab[] = []
    tabs.value.forEach((t, i) => {
      if (i >= idx || t.closable === false) newTabs.push(t)
    })
    tabs.value = newTabs
    if (activeKey.value !== key && newTabs.findIndex(t => t.key === activeKey.value) === -1) {
      activeKey.value = key
    }
    persist()
  }

  function closeRightTabs(key: string) {
    const idx = tabs.value.findIndex(t => t.key === key)
    if (idx === -1 || idx === tabs.value.length - 1) return
    const newTabs: Tab[] = []
    tabs.value.forEach((t, i) => {
      if (i <= idx || t.closable === false) newTabs.push(t)
    })
    tabs.value = newTabs
    if (activeKey.value !== key && newTabs.findIndex(t => t.key === activeKey.value) === -1) {
      activeKey.value = key
    }
    persist()
  }

  function moveTab(from: number, to: number) {
    if (from === to) return
    const moved = tabs.value.splice(from, 1)[0]
    tabs.value.splice(to, 0, moved)
    persist()
  }

  function setDragIndex(idx: number | null) { dragIndex.value = idx }
  function setDragOverIndex(idx: number | null) { dragOverIndex.value = idx }

  function persist() {
    saveToStorage(tabs.value, activeKey.value)
  }

  return {
    tabs, activeKey, activeTab, dragIndex, dragOverIndex,
    addTab, removeTab, setActiveKey,
    closeOtherTabs, closeAllTabs, closeLeftTabs, closeRightTabs,
    moveTab, setDragIndex, setDragOverIndex,
  }
})
```

### 3.2 `stores/theme.ts`

```typescript
import { defineStore } from 'pinia'
import { ref } from 'vue'

export type ThemeMode = 'light' | 'dark'
export type PrimaryColor = 'gold' | 'bamboo' | 'cinnabar'
export type FontSize = 'small' | 'standard' | 'large'
export type Radius = 'sharp' | 'rounded' | 'full'
export type Density = 'comfortable' | 'compact'
export type SiderWidth = 180 | 220 | 260

export interface UISettings {
  primary: PrimaryColor
  fontSize: FontSize
  radius: Radius
  density: Density
  siderWidth: SiderWidth
}

const DEFAULTS: UISettings = {
  primary: 'gold',
  fontSize: 'standard',
  radius: 'rounded',
  density: 'comfortable',
  siderWidth: 220,
}

const STORAGE_KEY = 'tea-ui-settings'

function loadSettings(): UISettings {
  try {
    const saved = localStorage.getItem(STORAGE_KEY)
    if (saved) return { ...DEFAULTS, ...JSON.parse(saved) }
  } catch {}
  return { ...DEFAULTS }
}

function loadThemeMode(): ThemeMode {
  try {
    return (localStorage.getItem('tea-theme') as ThemeMode) || 'light'
  } catch {
    return 'light'
  }
}

function applyToDOM(mode: ThemeMode, settings: UISettings) {
  const root = document.documentElement
  if (mode === 'dark') root.setAttribute('data-theme', 'dark')
  else root.removeAttribute('data-theme')

  root.setAttribute('data-primary', settings.primary)
  root.setAttribute('data-font-size', settings.fontSize)
  root.setAttribute('data-radius', settings.radius)
  root.setAttribute('data-density', settings.density)
  root.style.setProperty('--tea-sider-width', `${settings.siderWidth}px`)
}

export const useThemeStore = defineStore('theme', () => {
  const themeMode = ref<ThemeMode>(loadThemeMode())
  const settings = ref<UISettings>(loadSettings())
  const settingsOpen = ref(false)

  function setThemeMode(mode: ThemeMode) {
    themeMode.value = mode
    localStorage.setItem('tea-theme', mode)
    applyToDOM(mode, settings.value)
  }

  function toggleTheme() {
    setThemeMode(themeMode.value === 'dark' ? 'light' : 'dark')
  }

  function updateSettings(partial: Partial<UISettings>) {
    settings.value = { ...settings.value, ...partial }
    localStorage.setItem(STORAGE_KEY, JSON.stringify(settings.value))
    applyToDOM(themeMode.value, settings.value)
  }

  function resetSettings() {
    settings.value = { ...DEFAULTS }
    localStorage.removeItem(STORAGE_KEY)
    applyToDOM(themeMode.value, settings.value)
  }

  function openSettings() { settingsOpen.value = true }
  function closeSettings() { settingsOpen.value = false }

  // 初始化时立即应用
  if (typeof document !== 'undefined') {
    applyToDOM(themeMode.value, settings.value)
  }

  return {
    themeMode, settings, settingsOpen,
    setThemeMode, toggleTheme, updateSettings, resetSettings,
    openSettings, closeSettings,
  }
})
```

## 四、组件接口设计

### 4.1 `AppLayout.vue`
- 无 props
- 内部 use 路由 + tabs + theme
- 渲染 TitleBar + Sidebar + main + ThemeSettings

### 4.2 `TitleBar.vue`
- 无 props
- 渲染：品牌 + 折叠按钮 + TabBar + UserMenu
- 不接收 children

### 4.3 `TabBar.vue`
- 无 props
- 直接 useTabsStore
- 渲染标签列表 + 滚动 + 关闭按钮 + 右键菜单
- 处理 HTML5 DnD

### 4.4 `Sidebar.vue`
- props: `collapsed: boolean`
- emit: `update:collapsed`
- 内部：搜索框 + 菜单分组 + 折叠弹窗（createPortal）
- 菜单数据：硬编码（业务量小，单独文件 `sidebarMenus.ts`）

### 4.5 `ThemeSettings.vue`
- 无 props
- 从 useThemeStore 读取/写入设置
- 抽屉用 `n-drawer`

### 4.6 `UserMenu.vue`
- 无 props
- 主题切换、设置入口、退出登录
- 用 `n-dropdown`

## 六、关键交互流程

### 6.1 菜单点击 → 打开/激活标签
```
用户点击侧栏菜单项 { path: '/products', title: '商品档案' }
  → SidebarMenuItem @click
  → tabsStore.addTab({ key: '/products', title: '商品档案', closable: true })
  → router.push('/products')
  → AppLayout 监听 route.pathname → tabsStore.setActiveKey(pathname)
```

### 6.2 标签点击 → 切换页面
```
用户点击标签
  → TabBarItem @click
  → tabsStore.setActiveKey(tab.key)
  → router.push(tab.key)
  → router-view 重新渲染对应页面
```

### 6.3 标签关闭
```
用户点 X
  → tabsStore.removeTab(tab.key)
  → 如果关闭的是当前激活标签，自动激活下一个
  → AppLayout 监听 activeKey → router.push
```

### 6.4 防止循环
```typescript
// AppLayout.vue
const isUpdatingRef = ref(false)

watch(activeKey, (key) => {
  if (isUpdatingRef.value) return
  if (key !== route.path) {
    isUpdatingRef.value = true
    router.push(key)
  }
})

watch(() => route.path, (path) => {
  if (isUpdatingRef.value) {
    isUpdatingRef.value = false
    return
  }
  if (path !== tabsStore.activeKey) {
    tabsStore.setActiveKey(path)
  }
})
```

### 6.5 主题切换
```
用户点主题按钮
  → UserMenu @click "toggle theme"
  → themeStore.toggleTheme()
  → applyToDOM()
  → document.documentElement setAttribute('data-theme', 'dark'/'remove')
  → CSS 变量自动切换
```

## 七、与原代码的差异

| 原 | 新 | 原因 |
|----|----|------|
| `n-layout has-sider` | `<AppLayout>` 自实现 | Tauri 风格融合 |
| 顶部 `n-breadcrumb` | 删除 | 标签栏已承担 |
| 顶部 `current-time` | 删除 | 非业务必要 |
| `n-menu` 12 项平铺 | `Sidebar` 分 3 组 | 导航负担 |
| 无搜索 | 搜索框 | 12 项翻找费时 |
| 无主题切换 | 完整主题系统 | 用户需求 |
| 菜单硬编码 menuKeyToRoute | 菜单直接绑 path | 降低维护成本 |
| 无多任务 | 多标签页 | 核心需求 |

## 八、风险与对策

| 风险 | 影响 | 对策 |
|------|------|------|
| 多标签 + 路由循环 | 死循环白屏 | `isUpdatingRef` ref 防循环 |
| 大量标签溢出 | 标签栏变丑 | 滚动 + 遮罩 + 滚轮横向 |
| localStorage 损坏 | 白屏 | try/catch + 默认值 |
| 主题切换闪烁 | 用户体验差 | 在 main.ts 最顶部加载主题 |
| 现有页面用 n-card 等 | 无影响 | 主题只影响 token，组件内部样式自动跟随 |
